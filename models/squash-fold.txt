// Squash fold
// Fold up a triangular corner flap (hinged along p4-p5) so it stands
// perpendicular to the sheet, then crease it down its own centerline
// (apex p1 to the hinge midpoint p6). Pressing that centerline crease
// flat brings the apex all the way down onto the hinge midpoint,
// flattening the standing point.
// Note: a "real" squash fold opens two separate paper layers to either
// side; this simulator has no layers, so both layers here collapse onto
// the same spot, single ply, and only the core motion (apex descends flat
// onto the crease intersection) is shown.
define 200 200
textures
split s0 0.5
split s1 0.5
by2d p4 p5
// Mark the hinge midpoint the apex will fold down onto
split s6 0.5
// Stand the corner flap up
t 400 r s6 100 p1
// Crease down the centerline of the standing flap
by3d p1 p6
// Fold that centerline crease flat: the apex comes down onto the hinge midpoint
across3d p1 p6
t 500 r s12 180 p1
// Show
iao
t 1000 z 0.5 0 0
t 1000 fit
iao
t 1000 ty 360
